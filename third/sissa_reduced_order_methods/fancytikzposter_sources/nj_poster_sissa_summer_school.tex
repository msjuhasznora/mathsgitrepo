  \blocknode{\textcolor{gray}{The shallow domain concept}}
  {
  \begin{itemize}
    \item Natural, large-scale geophysical fluids share the shallow domain characteristics
    \item "almost" two-dimensional set $\to$ we can significantly simplify the model in the vertical direction
    \item a domain that is excessively greater horizontally than in the vertical dimension $\to$ leading to the classic hydrostatic approximation of the Navier-Stokes equations
  \end{itemize}
  
  \vspace{0.5cm}
  
  The main ideas and tools in this scenario:
  \begin{itemize}
    \item introduce the aspect ratio $\epsilon = \frac{\text{characteristic depth}}{\text{characteristic width}},$
    \item anisotropic viscosity hypothesis,
    \item rescaling, $x=x_1, y=x_2, z=\epsilon x_3.$
  \end{itemize}
  
  }

  Correspondingly, we rescale
  
  \begin{itemize}
    \item the kinematic terms: $v_3 = \epsilon u_3^{\epsilon},$
    \item the viscosity terms: $\nu_z = \epsilon ^ 2\nu_3,$
    \item the diffusivity terms: $K_x =  K_1, K_y = K_2, K_z = \epsilon ^ 2 K_3.$
  \end{itemize}

  \blocknode{\textcolor{gray}{Visualization of the convergence result: \textbf{a challenge}}}
  {
    \begin{itemize}
    
      \item In order to visualise the convergence result for now we use the finite element method, and
      \item the FEniCS platform.
      \item Space function choices for the velocity and the pressure --- this mixed finite element space is known as the \textbf{Taylor-Hood} elements and is a standard element pair for this kind of problem. In the original weak formulation the $p$ pressure function vanishes as a consequence of the divergence-free condition. In this discretised scenario we don't exactly have that constraint fulfilled, so we also use the pressure function in the weak formulation:
      
      \coloredbox{colorthree!50!}{
      
        \texttt{V = VectorFunctionSpace(mesh, "Lagrange", 2, dim = 3)} --- continuous piecewise quadratics
        
        \texttt{Q = FunctionSpace(mesh, "Lagrange", 1)} --- continuous piecewise linear functions
      }
      
      \item Challenge: \textbf{the FEM matrix is becoming badly conditioned} as the weight of $u^\epsilon_3$ is disappearing from the momentum equation --- as the shallowness parameter $\epsilon$ goes to zero.
      
      \item To obtain a discrete scheme in space and time we use the incremental pressure correction scheme --- \textbf{Chorin's 3-step method}
      
      \end{itemize}
      
    }

  {
IPCS - Incremental pressure correction scheme. How to update $(\mathbf{u}^n, p^n)$

to arrive to $(\mathbf{u}^{n+1}, p^{n+1})$ without increasing both quantities at the same time.

\begin{enumerate}
  \item Starting with $\mathbf{u}^n,$ calculate the so-called $\mathbf{u}^*$ tentative velocity using only the $p^n$ pressure value from the previous time step.
  \item Updating the pressure, step from $p^n$ to $p^{n+1}$ based on $\mathbf{u}^*.$ (N-S@($\mathbf{u}^*,$ $p^n$) - N-S@($\mathbf{u}^{n+1},$ $p^{n+1}$))
  \item Calculating the definitive $\mathbf{u}^{n+1}$ velocity based on $p^{n+1}$ and $\mathbf{u}^*.$
\end{enumerate}
  }

    \blocknode{\textcolor{gray}{FEniCS code}}{
    
      \begin{itemize}
      
      \item STEP1 --- tentative velocity calculation ignoring the pressure change
      
        \coloredbox{colorthree!50!}{
       
      
        \texttt{ $F1\_anisotropic = $ }
      
        \texttt{ $ (1/k)*\big( (u1 - u\_prev1)*v1 + (u2 - u\_prev2)*v2 + \epsilon*\epsilon*\mathbf{(u3 - u\_prev3)*v3} \big) * dx + $ }
      
         \texttt{ $ + inner(u\_prev, grad(u\_prev1)) * v1 * dx + inner(u\_prev, grad(u\_prev2)) * v2 * dx + \epsilon*\epsilon*inner(u\_prev, grad(u\_prev3)) * v3 * dx $}
        
         \texttt{ $ - \alpha * u\_prev2 * v1 * dx + \alpha * u\_prev1 * v2 * dx + inner(grad(u1),grad(v1)) * dx + inner(grad(u2),grad(v2)) * dx + \epsilon*\epsilon*\mathbf{inner(grad(u3),grad(v3))} * dx + \epsilon * \beta * u\_prev3 * v1 * dx - \epsilon * \beta * u\_prev1 * v3 * dx - inner(f, v) * dx - inner(\theta, v) * ds(1) $ }
       
          \texttt{ $ a1 = lhs(F1\_anisotropic), L1 = rhs(F1\_anisotropic) $}
         
       }
    
      \item STEP2 --- updating the pressure in a separate step
      
        \coloredbox{colorthree!50!}{
      
          \texttt{  $ a2 = inner(grad(p), grad(q))*dx, L2 = - (1/k)*\text{div}(u\_next)*q*dx $ }
        
        }
    
      \item STEP3 --- updating the velocity knowing the new pressure value
      
        \coloredbox{colorthree!50!}{

          \texttt{ $ a3 = inner(u, v)*dx, L3 = inner(u\_next, v)*dx - k*inner(grad(p\_next), v)*dx $ }
        
        }
      
    \end{itemize}
  }
