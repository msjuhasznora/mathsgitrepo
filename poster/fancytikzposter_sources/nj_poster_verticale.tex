  \blocknode{The shallow domain concept}
  {
  \begin{itemize}
    \item Natural, large-scale geophysical fluids share the shallow domain characteristics
    \item "almost" two-dimensional set $\to$ we can significantly simplify the model in the vertical direction
    \item a domain that is excessively greater horizontally than in the vertical dimension $\to$ leading to the classic hydrostatic approximation of the Navier-Stokes equations
  \end{itemize}
  
  \vspace{0.5cm}

  \begin{tikzfigure}[A local slice of the atmosphere. Picture credit: mountwashington.org]
    \begin{tikzpicture}
      \node[inner sep=0pt] (atmospherelayers) at (0,0) {\includegraphics[width=.25\textwidth]{atmosphere.jpg}};
      \end{tikzpicture}
  \end{tikzfigure}
  
  The main ideas and tools in this scenario:
  \begin{itemize}
    \item introduce the aspect ratio $\epsilon = \frac{\text{characteristic depth}}{\text{characteristic width}},$
    \item anisotropic viscosity hypothesis,
    \item rescaling, $x=x_1, y=x_2, z=\epsilon x_3.$
  \end{itemize}
  
  }

  {\small
  Correspondingly, we rescale
  
  \begin{itemize}
    \item the kinematic terms: $v_3 = \epsilon u_3^{\epsilon},$
    \item the viscosity terms: $\nu_z = \epsilon ^ 2\nu_3,$
    \item the diffusivity terms: $K_x = \epsilon K_1, K_y = K_2, K_z = \epsilon ^ 2 K_3.$
  \end{itemize}
  }
  
    %%%%%%%%%% ------------------------------------------ %%%%%%%%%%
  \blocknode{Modelling - challenges and choices}
  {
  \begin{itemize}
    \item Coordinate system $\to$ we fix it in a way so that $x$ and $y$ match the downwind and crosswind direction, respectively (while $z$ represents the vertical dimension).
    
    Natural assumption in the downwind direction: diffusion is negligible compared to downwind transport, i.e.
     
    \[ \abs{u_1 \partial_{x_1} C} \gg \abs{ K_x \frac{\partial ^2 C}{\partial x_1 ^ 2} },\]
     
    so we can take advantage of the additional scaling equation $K_x = \epsilon K_1$.
     
    \item Domain $\to$ we use locally acceptable Cartesian coordinates --- this allows us the ignore large-scale effects such as the curvature of the Earth.
     
      \begin{tikzfigure}[A local slice of the atmosphere]
        \begin{tikzpicture}
      
          \node[inner sep=0pt] (domain) at (0,0) {\includegraphics[width=.15\textwidth]{domain.jpg}};
      
        \end{tikzpicture}
      \end{tikzfigure}
  
    Challenges: local area model $\to$ open boundary conditions on the lateral boundary. We use a fix zero approach --- the concentration being set to zero actually stands for a constant concentration with the background flow being removed.
    
    \item Source function of the concentration $\to$ Gaussian impulse approach. We use a parametrized nascent delta function to describe the source term, with the parameter being $\epsilon$. We have $S_\epsilon (t, t_s, x, x_s) = s(t, t_s) \delta_\epsilon (x - x_s),$ with the activation term $s(t,t_s)= IH(t - t_s)$ where $H(\cdot)$ is the Heaviside step function. In the limit model we naturally arrive to the delta distribution as a source.
    
  \end{itemize}
  }

    \coloredbox{colorthree!50!}{

    \[ \partial_t u_2^\epsilon + \bold{u}^\epsilon \cdot \nabla u_2^\epsilon -\Delta_\nu u_2^\epsilon + \alpha u_1^\epsilon + \partial_2 p^\epsilon = 0 \text{ in } \Omega \times (0,T) \]
    
    \[\epsilon^2 \{  \partial_t u_3^\epsilon + \bold{u}^\epsilon \cdot \nabla u_3^\epsilon -\Delta_\nu u_3^{\epsilon} \} - \epsilon \beta u_1^\epsilon + \partial_3 p^\epsilon = 0 \text{ in } \Omega \times (0,T) \]
    
    \[ \nabla \cdot \bold{u}^\epsilon= 0 \text{ in } \Omega \times (0,T) \]
    
    \[ \partial_t C^\epsilon + \bold{u}^\epsilon \cdot \nabla C^\epsilon = \epsilon K_1 \frac{\partial ^2 C^\epsilon}{\partial x_1 ^ 2} + K_2 \frac{\partial ^2 C^\epsilon}{\partial x_2 ^ 2} + K_3 \frac{\partial ^2 C^\epsilon}{\partial x_3 ^ 2} + S_\epsilon \text{ in } \Omega \times (0,T) \]
    
    \[ \bold{u}^\epsilon = 0 \text{ on } \Gamma \times (0,T) \] 
    
    \[ \bold{u}^\epsilon (\cdot, t = 0) = \bold{u}_0, C^\epsilon (\cdot, t=0) = C^\epsilon_0  \text{ in } \Omega \]

    \[ C^\epsilon = 0 \text{ on } ( \Gamma_U \cup \Gamma_L) \times (0,T), \partial_2 C^\epsilon = \partial_3 C^\epsilon = 0 \text{ on } \Gamma_G \times (0,T), \partial_1 C^\epsilon \in L^2 L^2 (0,T; \Gamma_G) \]

  \blocknode{The limit model}
  { \small In the vertical dimension we arrive to the hydrostatic approximation.
    \vspace{0.5cm}
    \coloredbox{colorthree!50!}{
    
    \[ \partial_t u_1 + \bold{u} \cdot \nabla u_1 - \Delta_\nu u_1 -\alpha u_2 + \partial_1 p = 0 \text{ in } \Omega \times (0,T) \]
    
    \[ \partial_t u_2 + \bold{u} \cdot \nabla u_2 - \Delta_\nu u_2 -\alpha u_1 + \partial_2 p = 0 \text{ in } \Omega \times (0,T) \]
    
    \[ \partial_3 p = 0 \text{ in } \Omega \times (0,T) \]
    
    \[ \nabla \cdot \bold{u} = 0 \text{ in } \Omega \times (0,T) \]
    
    \[ \partial_t C + \bold{u} \cdot \nabla C = K_2 \frac{\partial ^2 C}{\partial x_2 ^ 2} + K_3 \frac{\partial ^2 C}{\partial x_3 ^ 2} + S \text{ in } \Omega \times (0,T)\]
    
    \[ u_1 = u_2 = u_3 n_3 = 0 \text{ on } \Gamma \times (0,T) \]
    
    \[ u_i (\cdot, t=0) = u_{0 i}, C (\cdot, t=0) = C_0 \text{ in } \Omega, \text{$i$ = 1,2} \]
    
    \[ C = 0 \text{ on } ( \Gamma_U \cup \Gamma_L) \times (0,T), \partial_2 C = \partial_3 C = 0 \text{ on } \Gamma_G \times (0,T), \partial_1 C \in L^2 L^2 (0,T; \Gamma_G) \]
    }
  }

  \blocknode {A priori estimates, weak convergence}
  {
  
    \small The main steps for proving the weak convergence are the following.
    \begin{itemize}
      \item Energy inequality.
    \end{itemize}
      
    \coloredbox{colorthree!50!}{
    
    \[
    \begin{split}  
    & \frac{1}{2} (\norm{\mathbf{u}_H^\epsilon (t)}_{L^2}^2 + \epsilon ^ 2 \norm{u_3^\epsilon (t)}_{L^2}^2 + \norm{C^\epsilon (t)}_{L^2}^2) + \int_0 ^ t \{    \norm{\nabla_\nu \mathbf{u}_H^\epsilon}_{L^2}^2 + \epsilon ^ 2 \norm{\nabla_\nu u_3^\epsilon}_{L^2}^2  \} \diff \tau + \\
    & \int_0 ^ t \{(\epsilon K_1 - \gamma \frac{\epsilon}{2} K_1 c_T (c_S + 1)) \norm{\partial_1 C^\epsilon}_{L^2}^2  + (K_2 - \gamma \frac{\epsilon}{2} K_1 c_T (c_S + 1)) \norm{\partial_2 C^\epsilon}_{L^2}^2 + \\
    & + (K_3 - \gamma \frac{\epsilon}{2} K_1 c_T (c_S + 1)) \norm{\partial_3 C^\epsilon}_{L^2}^2 \} \diff \tau \leq \frac{1}{\gamma} K_1 \frac{\epsilon}{2} \int_0 ^ t \norm{\partial_1 C^\epsilon}^2 _{L^2 (\Gamma_G)} + \\
    & + \frac{1}{2} (\norm{\mathbf{u}_H^\epsilon (0)}_{L^2}^2 + \epsilon ^ 2 \norm{u_3^\epsilon (0)}_{L^2}^2 + \norm{C^\epsilon (0)}_{L^2}^2) + \int_0^t (S^\epsilon,C^\epsilon) \diff \tau.
    \end{split}
    \]
    
    }
      
    \begin{itemize}  
      \item A priori estimates: $L_t^\infty L_x^2$ boundedness for $C^\epsilon;$ $L_t^2 L_x^2$ boundedness for $\sqrt{\epsilon} \partial_1 C^\epsilon, \partial_2 C^\epsilon, \partial_3 C^\epsilon.$
      \item Weak convergence of the linear terms.
      \item Weak convergence of nonlinear terms.
      \begin{itemize}
        \item Nonlinear terms not including the $C^\epsilon$ term - divergence free condition
        \item The term $(\mathbf{u}^\epsilon C^\epsilon, \nabla \tilde{C}).$
      \end{itemize}
    \end{itemize}
  }

  {
    We need a compactness result to obtain $u_i^\epsilon C^\epsilon \rightharpoonup  u_i C.$
    
    For this we show that
    
    \begin{itemize}
    
      \item we have a bound for the time derivative of the concentration:
      
      \[  \frac{\partial C^\epsilon}{\partial t} = - u^\epsilon \nabla C^\epsilon + \sqrt{\epsilon} K_1 \partial_1 \partial_1 \sqrt{\epsilon} C^\epsilon + K_2 \partial_2 \partial_2 C^\epsilon + K_3 \partial_3 \partial_3 C^\epsilon -S^\epsilon, \]
      
      Eventually we have $\frac{\partial C}{\partial t} \in L^1 W^{-1,1},$
      
      \item we have an equicontinuity for velocity: $\norm{u_i^n - u_i^n (\cdot + \xi,t)}_{L^2 L^2} \to 0$ as $\xi \to 0.$
    
    \end{itemize}
    
    Combining these results allow us to apply a compactness theorem that provides the convergence.
  }
